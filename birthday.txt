        function typeWriter() {

            if (index < message.length) {

                const char =
                    message.charAt(index);


                if (char === "\n") {

                    typewriterElement.appendChild(
                        document.createElement("br")
                    );

                } else {

                    typewriterElement.appendChild(
                        document.createTextNode(char)
                    );
                }


                index++;


                /*
                    Tự cuộn thư xuống trong lúc
                    typewriter đang chạy.
                */

                scrollContainer.scrollTop =
                    scrollContainer.scrollHeight;


                setTimeout(
                    typeWriter,
                    32
                );

            } else {

                /*
                    Đọc xong thư -> chờ một chút
                    rồi chuyển sang bánh.
                */

                setTimeout(
                    switchToCakeScreen,
                    2500
                );
            }
        }

        function openLetter() {

            if (isOpened) {
                return;
            }


            isOpened = true;


            const envelope =
                document.getElementById(
                    "envelopeWrapper"
                );


            envelope.classList.add(
                "open"
            );


            setTimeout(() => {

                if (!isTyping) {

                    isTyping = true;

                    typeWriter();

            }, 1100);
        }

        let fireworksRunning = false;


        function startFireworks() {

            if (fireworksRunning) {
                return;
            }


            fireworksRunning = true;


            fireworksContainer.classList.add(
                "active"
            );


            /*
                Bắn pháo hoa liên tục trong
                khoảng 8 giây.
            */

            const fireworksInterval =
                setInterval(() => {

                    createFirework();

                }, 380);


            /*
                Bắn một số quả ngay lập tức
                để màn hình không bị trống.
            */

            createFirework();

            setTimeout(
                createFirework,
                150
            );

                createFirework,
                300
            );


            /*
                Sau 8 giây dừng việc tạo
                pháo hoa mới.
            */

            setTimeout(() => {

                clearInterval(
                    fireworksInterval
                );


                /*
                    Cho những tia cuối cùng
                    bay hết rồi mới dọn DOM.
                */

                setTimeout(() => {

                    fireworksContainer.innerHTML =
                        "";

                    fireworksContainer.classList.remove(
                        "active"
                    );

                    fireworksRunning =
                        false;

                }, 1800);

            }, 8000);
        }

        function createFirework() {

            const x =
                Math.random() *
                window.innerWidth;


            /*
                Vị trí nổ nằm ở khoảng
                15% -> 55% chiều cao màn hình.
            */

            const explosionY =
                window.innerHeight *
                (
                    0.15 +
                    Math.random() *
                    0.45
                );


            /*
                Tạo lõi pháo hoa.
            */

            const core =
                document.createElement(
                    "div"
                );


            core.className =
                "firework-core";


            core.style.left =
                x + "px";


            core.style.top =
                explosionY + "px";


            fireworksContainer.appendChild(
                core
            );


            /*
                Tạo khoảng 28-42 tia
                cho mỗi quả pháo.
            */

            const particleCount =
                28 +
                Math.floor(
                    Math.random() * 15
                );


            for (
                let i = 0;
                i < particleCount;
                i++
            ) {

                const particle =
                    document.createElement(
                        "div"
                    );


                particle.className =
                    "firework-particle";


                particle.style.left =
                    x + "px";


                particle.style.top =
                    explosionY + "px";


                /*
                    Góc tia.
                */

                const angle =
                    (
                        Math.PI * 2 *
                        i
                    ) /
                    particleCount;


                /*
                    Bán kính ngẫu nhiên
                    tạo cảm giác tự nhiên.
                */

                const distance =
                    70 +
                    Math.random() *
                    140;


                const moveX =
                    Math.cos(angle) *
                    distance;


                const moveY =
                    Math.sin(angle) *
                    distance;


                particle.style.setProperty(
                    "--x",
                    moveX + "px"
                );


                particle.style.setProperty(
                    "--y",
                    moveY + "px"
                );


                /*
                    Mỗi tia bắt đầu hơi lệch
                    thời gian một chút.
                */

                particle.style.animationDelay =
                    (
                        Math.random() *
                        0.08
                    ) +
                    "s";


                fireworksContainer.appendChild(
                    particle
                );


                setTimeout(() => {

                    particle.remove();

                }, 1500);
            }


            setTimeout(() => {

                core.remove();

            }, 600);
        }
