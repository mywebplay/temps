		// Thêm 1 trang ảnh vào PDF theo kiểu "COVER": phủ kín trang A4, căn giữa
		function addCoverImage(pdf, imgData, ratio, alias, compression, isFirstPage) {
		    const pageWidth = pdf.internal.pageSize.getWidth();
		    const pageHeight = pdf.internal.pageSize.getHeight();
		    const pageRatio = pageWidth / pageHeight;
		
		    let imgWidth, imgHeight;
		    if (ratio > pageRatio) {
		        imgHeight = pageHeight;
		        imgWidth = pageHeight * ratio;
		    } else {
		        imgWidth = pageWidth;
		        imgHeight = pageWidth / ratio;
		    }
		    const x = (pageWidth - imgWidth) / 2;
		    const y = (pageHeight - imgHeight) / 2;
		
		    if (!isFirstPage) pdf.addPage();
		    pdf.addImage(imgData, "JPEG", x, y, imgWidth, imgHeight, alias, compression);
		}

		// canvas -> { data: Uint8Array (JPEG), ratio } ; giải phóng canvas sau khi dùng
		async function canvasToJpeg(canvas, quality) {
		    if (!canvas || canvas.width === 0 || canvas.height === 0) return null;
		    const ratio = canvas.width / canvas.height;
		    const cw = canvas.width, ch = canvas.height;
		    // toBlob mã hóa bất đồng bộ, nhanh và nhẹ hơn toDataURL (base64)
		    const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", quality));
		    canvas.width = canvas.height = 0; // trả bộ nhớ sớm
		    if (!blob) return null;
		    return { data: new Uint8Array(await blob.arrayBuffer()), ratio: ratio, w: cw, h: ch };
		}

		// Ném lỗi nếu canvas bị "tainted" (ảnh khác nguồn) -> người gọi sẽ rơi về tầng sau.
		function isBlankCanvas(canvas) {
		    const S = 24;
		    const small = document.createElement("canvas");
		    small.width = small.height = S;
		    const ctx = small.getContext("2d", { willReadFrequently: true });
		    ctx.drawImage(canvas, 0, 0, S, S);
		    const d = ctx.getImageData(0, 0, S, S).data;
		    for (let i = 0; i < d.length; i += 4) {
		        if (d[i + 3] < 10) continue; // pixel trong suốt
		        if (d[i] < 250 || d[i + 1] < 250 || d[i + 2] < 250) return false;
		    }
		    return true;
		}
		
		// Tầng 1: trang chỉ là 1 ảnh/canvas phủ gần kín -> lấy thẳng, không cần "chụp" DOM.
		// Trả null nếu trang không đủ điều kiện (để không làm mất nội dung).
		async function tryDirectImage(div, opt) {
		    if (div.textContent.trim() !== "") return null;                                // có chữ -> không dùng
		    if (div.querySelector("svg, video, iframe, object, embed")) return null;       // có thành phần khác
		    const medias = div.querySelectorAll("img, canvas");
		    if (medias.length !== 1) return null;
		    const el = medias[0];
		
		    // Có ảnh nền ở đâu đó thì nội dung đó sẽ bị mất -> không dùng
		    const all = [div, ...div.querySelectorAll("*")];
		    if (all.some((n) => getComputedStyle(n).backgroundImage !== "none")) return null;
		
		    const d = div.getBoundingClientRect();
		    const r = el.getBoundingClientRect();
		    if (d.width < 1 || d.height < 1) return null;
		    if (r.width * r.height < 0.9 * d.width * d.height) return null;                // ảnh không phủ gần kín trang
		
		    if (el.tagName === "IMG") {
		        if (!(el.complete && el.naturalWidth > 0)) return null;
		        const fit = getComputedStyle(el).objectFit;
		        if (fit && fit !== "fill") return null;
		    } else if (el.width === 0 || el.height === 0) {
		        return null;
		    }
		
		    const c = document.createElement("canvas");
		    c.width = Math.round(opt.targetWidth);
		    c.height = Math.max(1, Math.round((opt.targetWidth * d.height) / d.width));
		    const sx = c.width / d.width;
		    const ctx = c.getContext("2d");
		    ctx.fillStyle = "#ffffff";
		    ctx.fillRect(0, 0, c.width, c.height);
		    ctx.imageSmoothingQuality = "high";
		    ctx.drawImage(el, (r.left - d.left) * sx, (r.top - d.top) * sx, r.width * sx, r.height * sx);
		
		    // Ném SecurityError nếu ảnh khác nguồn không có CORS -> người gọi sẽ bắt lại
		    return await canvasToJpeg(c, opt.quality);
		}
		
		// Chạy promise với giới hạn thời gian; quá hạn thì ném lỗi
		function withTimeout(promise, ms, label) {
		    let timer;
		    const timeout = new Promise((_, reject) => {
		        timer = setTimeout(() => reject(new Error("quá " + ms + "ms: " + label)), ms);
		    });
		    return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
		}
		
		// Render 1 trang theo thứ tự: direct -> foreignObject -> html2canvas thường
		// state: cờ dùng chung, tầng nào đã lỗi thì các trang sau bỏ qua luôn
		async function renderPageFast(div, opt, state) {
		    const tag = "[trang " + div.getAttribute("data-page-index") + "]";
		    const lap = (name, t0) => console.log(tag, name, Math.round(performance.now() - t0) + "ms");
		
		    // --- Tầng 1 ---
		    if (state.direct) {
		        const t0 = performance.now();
		        try {
		            const r = await tryDirectImage(div, opt);
		            lap(r ? "direct OK" : "direct bỏ qua", t0);
		            if (r) return { ...r, method: "direct" };
		        } catch (e) {
		            lap("direct lỗi (" + e.message + ")", t0);
		            state.direct = false;
		        }
		    }
		
		    const w = div.offsetWidth || div.scrollWidth || 1;
		    const scale = Math.min(Math.max(opt.targetWidth / w, opt.minScale), opt.maxScale);
		    const baseOpts = {
		        scale: scale,
		        useCORS: true,
		        backgroundColor: "#ffffff",
		        logging: false,
		        windowWidth: div.scrollWidth,
		        windowHeight: div.scrollHeight,
		    };
		
		    // --- Tầng 2 (có giới hạn thời gian) ---
		    if (state.foreign) {
		        const t0 = performance.now();
		        try {
		            const canvas = await withTimeout(
		                html2canvas(div, { ...baseOpts, foreignObjectRendering: true }),
		                opt.tierTimeoutMs || 4000,
		                "foreignObject"
		            );
		            if (!isBlankCanvas(canvas)) {
		                const r = await canvasToJpeg(canvas, opt.quality);
		                if (r) {
		                    lap("foreignObject OK", t0);
		                    return { ...r, method: "foreignObject" };
		                }
		            }
		            lap("foreignObject ra trang trắng", t0);
		            state.foreign = false;
		        } catch (e) {
		            lap("foreignObject lỗi (" + e.message + ")", t0);
		            state.foreign = false;
		        }
		    }
		
		    // --- Tầng 3 ---
		    const t1 = performance.now();
		    const canvas = await html2canvas(div, { ...baseOpts, foreignObjectRendering: false });
		    lap("html2canvas", t1);
		    const t2 = performance.now();
		    const r = await canvasToJpeg(canvas, opt.quality);
		    lap("toBlob", t2);
		    return r ? { ...r, method: "standard" } : null;
		}

		        const tStart = performance.now();

		            for (let i = 0; i < total; i++) {
		                const t0 = performance.now();
		
		                pages[i].style.display = savedDisplay[i];
		                let r = null;
		                try {
		                    r = await renderPageFast(pages[i], opt, state);
		                } finally {
		                    pages[i].style.display = "none";
		                }
		                const dt = performance.now() - t0;
		
		                if (i === 0) showProgressOverlay();
		                if (r) {
		                    results[i] = r;
		                    methodCount[r.method] = (methodCount[r.method] || 0) + 1;
		                }
		
		                // Sau trang đầu: nếu ước tính quá lâu thì đề xuất chuyển sang chế độ in
		                if (i === 0 && opt.suggestPrint && r && r.method !== "direct" && dt * total > opt.slowEstimateMs) {
		                    const sec = Math.round((dt * total) / 1000);
		                    alert(
		                        "Xử lý chụp ảnh PDF đối với tài liệu này có thể khá chậm (ước tính khoảng " + sec + " giây).\n\n" +
		                        "Bấm [OK] để tiếp tục xử lý..."
		                    );
		                }
		
		                updateProgress(Math.round(((i + 1) / total) * 100));
		                console.log(`Trang ${i}: ${r ? r.method : "lỗi"} - ${Math.round(dt)}ms`);
		                await sleep(0);
		            }
		        } finally {
		            // Luôn trả lại trạng thái hiển thị ban đầu của các trang
		            pages.forEach((p, i) => (p.style.display = savedDisplay[i]));
		        }
		        console.log("Cách render mỗi trang:", methodCount, "| Tổng:", Math.round(performance.now() - tStart), "ms");
		
		        // Ghép PDF theo đúng thứ tự trang
		        let added = 0;
		        for (let i = 0; i < total; i++) {
		            const r = results[i];
		            if (!r) continue;
		            // alias duy nhất giúp jsPDF bỏ qua bước băm dữ liệu ảnh để tìm ảnh trùng
		            addCoverImage(pdf, r.data, r.ratio, "p" + i, "NONE", added === 0);
		            results[i] = null;
		            added++;
		        }
		        if (added === 0) throw new Error("Không render được trang nào");
		
		        downloadPdf(pdf);
		    });
		
		    if (needPrint) return exportPdfPrint();
		}

		// ---------- Ghi file PDF thủ công ----------
		const _enc = new TextEncoder();
		const _hex2 = (n) => n.toString(16).padStart(2, "0");
		const _hex4 = (n) => n.toString(16).padStart(4, "0");
		const _f2 = (n) => String(Math.round(n * 100) / 100);
		
		// Mã 1 byte dùng cho mỗi ký tự (tránh vùng điều khiển 0-32 và 127-160)
		const TEXT_CODES = (() => {
		    const a = [];
		    for (let i = 33; i <= 126; i++) a.push(i);
		    for (let i = 161; i <= 255; i++) a.push(i);
		    return a;
		})();
		
		function _utf16hex(ch) {
		    const cp = ch.codePointAt(0);
		    if (cp > 0xFFFF) {
		        const c = cp - 0x10000;
		        return _hex4(0xD800 + (c >> 10)) + _hex4(0xDC00 + (c & 0x3FF));
		    }
		    return _hex4(cp);
		}
		
		// Gán mỗi ký tự một (font, mã). Mỗi font tối đa ~189 ký tự khác nhau, hết thì mở font mới.
		class TextCodec {
		    constructor() { this.map = new Map(); this.fonts = []; }
		    encodeRun(text) {
		        const segs = [];
		        let cur = null;
		        for (const ch of text) {
		            let e = this.map.get(ch);
		            if (!e) {
		                let f = this.fonts[this.fonts.length - 1];
		                if (!f || f.chars.length >= TEXT_CODES.length) { f = { chars: [] }; this.fonts.push(f); }
		                e = { font: this.fonts.length - 1, code: TEXT_CODES[f.chars.length] };
		                f.chars.push(ch);
		                this.map.set(ch, e);
		            }
		            if (!cur || cur.font !== e.font) { cur = { font: e.font, hex: "" }; segs.push(cur); }
		            cur.hex += _hex2(e.code);
		        }
		        return segs;
		    }
		}
		
		function _buildToUnicode(chars) {
		    let s = "/CIDInit /ProcSet findresource begin\n12 dict begin\nbegincmap\n" +
		        "/CIDSystemInfo << /Registry (Adobe) /Ordering (UCS) /Supplement 0 >> def\n" +
		        "/CMapName /Adobe-Identity-UCS def\n/CMapType 2 def\n" +
		        "1 begincodespacerange\n<00> <FF>\nendcodespacerange\n";
		    for (let i = 0; i < chars.length; i += 100) {
		        const part = chars.slice(i, i + 100);
		        s += part.length + " beginbfchar\n";
		        part.forEach((ch, j) => { s += "<" + _hex2(TEXT_CODES[i + j]) + "> <" + _utf16hex(ch) + ">\n"; });
		        s += "endbfchar\n";
		    }
		    s += "endcmap\nCMapName currentdict /CMap defineresource pop\nend\nend\n";
		    return _enc.encode(s);
		}
		
		class PdfWriter {
		    constructor() { this.chunks = []; this.offsets = []; this.size = 0; this.nextId = 1; }
		    reserve() { return this.nextId++; }
		    _push(u8) { this.chunks.push(u8); this.size += u8.length; }
		    _str(s) { this._push(_enc.encode(s)); }
		    begin() { this._push(new Uint8Array([0x25, 0x50, 0x44, 0x46, 0x2D, 0x31, 0x2E, 0x34, 0x0A, 0x25, 0xE2, 0xE3, 0xCF, 0xD3, 0x0A])); }
		    obj(id, body, stream) {
		        this.offsets[id] = this.size;
		        if (stream) {
		            this._str(id + " 0 obj\n<< " + body + " /Length " + stream.length + " >>\nstream\n");
		            this._push(stream);
		            this._str("\nendstream\nendobj\n");
		        } else {
		            this._str(id + " 0 obj\n" + body + "\nendobj\n");
		        }
		    }
		    finish(rootId) {
		        const xrefPos = this.size;
		        const n = this.nextId;
		        let x = "xref\n0 " + n + "\n0000000000 65535 f \n";
		        for (let i = 1; i < n; i++) x += String(this.offsets[i]).padStart(10, "0") + " 00000 n \n";
		        x += "trailer\n<< /Size " + n + " /Root " + rootId + " 0 R >>\nstartxref\n" + xrefPos + "\n%%EOF\n";
		        this._str(x);
		        return new Blob(this.chunks, { type: "application/pdf" });
		    }
		}
		
		// items: [{ jpg: {data, w, h}, runs: [{text,x,y,w,h}], pageW, pageH }]  (toạ độ chữ tính theo px của trang)
		function buildTextPdf(items, landscape) {
		    const PW = landscape ? 841.89 : 595.28;
		    const PH = landscape ? 595.28 : 841.89;
		
		    // 1) Mã hóa toàn bộ chữ trước để biết cần bao nhiêu font
		    const codec = new TextCodec();
		    const encoded = items.map((it) => it.runs.map((r) => ({ r, segs: codec.encodeRun(r.text), n: Array.from(r.text).length })));
		
		    const w = new PdfWriter();
		    w.begin();
		    const catalogId = w.reserve();
		    const pagesId = w.reserve();
		
		    // 2) Font (không nhúng glyph, chữ vô hình nên không cần), mọi ký tự rộng đúng 1 em
		    const widths = "[" + new Array(256).fill(1000).join(" ") + "]";
		    const fontIds = codec.fonts.map((f) => ({ tuId: w.reserve(), fid: w.reserve(), f }));
		    fontIds.forEach(({ tuId, fid, f }) => {
		        w.obj(tuId, "", _buildToUnicode(f.chars));
		        w.obj(fid, "<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica /Encoding /WinAnsiEncoding" +
		            " /FirstChar 0 /LastChar 255 /Widths " + widths + " /ToUnicode " + tuId + " 0 R >>");
		    });
		    const fontRes = fontIds.map((x, i) => "/F" + (i + 1) + " " + x.fid + " 0 R").join(" ");
		
		    // 3) Từng trang
		    const kids = [];
		    items.forEach((it, pi) => {
		        const imgId = w.reserve();
		        const contId = w.reserve();
		        const pageId = w.reserve();
		        kids.push(pageId + " 0 R");
		
		        // "COVER": ảnh phủ kín trang, căn giữa (giống chế độ jsPDF)
		        const ratio = it.pageW / it.pageH;
		        let iw, ih;
		        if (ratio > PW / PH) { ih = PH; iw = PH * ratio; } else { iw = PW; ih = PW / ratio; }
		        const ix = (PW - iw) / 2;
		        const iyTop = (PH - ih) / 2;
		        const sx = iw / it.pageW;
		        const sy = ih / it.pageH;
		
		        let c = "q " + _f2(iw) + " 0 0 " + _f2(ih) + " " + _f2(ix) + " " + _f2(PH - iyTop - ih) + " cm /Im0 Do Q\n";
		        c += "BT\n3 Tr\n"; // 3 Tr = chữ vô hình
		        for (const { r, segs, n } of encoded[pi]) {
		            if (!n) continue;
		            const fs = Math.max(r.h * sy, 1);
		            const tz = Math.min(Math.max(((r.w * sx) / (n * fs)) * 100, 1), 20000);
		            const px = ix + r.x * sx;
		            const py = PH - (iyTop + (r.y + r.h * 0.8) * sy);
		            segs.forEach((sg, si) => {
		                c += "/F" + (sg.font + 1) + " " + _f2(fs) + " Tf ";
		                if (si === 0) c += _f2(tz) + " Tz 1 0 0 1 " + _f2(px) + " " + _f2(py) + " Tm ";
		                c += "<" + sg.hex + "> Tj\n";
		            });
		        }
		        c += "ET\n";
		
		        w.obj(imgId, "/Type /XObject /Subtype /Image /Width " + it.jpg.w + " /Height " + it.jpg.h +
		            " /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode", it.jpg.data);
		        w.obj(contId, "", _enc.encode(c));
		        w.obj(pageId, "<< /Type /Page /Parent " + pagesId + " 0 R /MediaBox [0 0 " + _f2(PW) + " " + _f2(PH) + "]" +
		            " /Resources << /XObject << /Im0 " + imgId + " 0 R >> /Font << " + fontRes + " >> >>" +
		            " /Contents " + contId + " 0 R >>");
		    });
		
		    w.obj(pagesId, "<< /Type /Pages /Kids [" + kids.join(" ") + "] /Count " + kids.length + " >>");
		    w.obj(catalogId, "<< /Type /Catalog /Pages " + pagesId + " 0 R >>");
		    return w.finish(catalogId);
		}

		function _colorAlpha(c) {
		    if (!c || c === "transparent") return 0;
		    const m = c.match(/rgba?\(([^)]+)\)/);
		    if (!m) return 1;
		    const p = m[1].split(/[ ,\/]+/).filter(Boolean);
		    return p.length >= 4 ? parseFloat(p[3]) : 1;
		}
		
		// Có thứ gì mà bộ vẽ nhanh không tái hiện được không?
		function _liteUnsupported(el, cs) {
		    const op = parseFloat(cs.opacity);
		    if (op > 0 && op < 1) return true;
		    if (cs.backgroundImage !== "none") return true;
		    for (const s of ["Top", "Right", "Bottom", "Left"]) {
		        if (cs["border" + s + "Style"] !== "none" && parseFloat(cs["border" + s + "Width"]) > 0) return true;
		    }
		    if (cs.boxShadow !== "none" || cs.filter !== "none" || cs.mixBlendMode !== "normal" || cs.clipPath !== "none") return true;
		    if (cs.textDecorationLine !== "none") return true;
		    if (cs.writingMode !== "horizontal-tb") return true;
		    if (cs.transform !== "none") {
		        const m = cs.transform.match(/^matrix\(([^)]+)\)$/);
		        if (!m) return true;
		        const v = m[1].split(",").map(Number);
		        if (Math.abs(v[1]) > 1e-3 || Math.abs(v[2]) > 1e-3) return true; // xoay / nghiêng
		    }
		    for (const p of ["::before", "::after"]) {
		        const ct = getComputedStyle(el, p).content;
		        if (ct && ct !== "none" && ct !== "normal" && ct !== '""' && ct !== "''") return true;
		    }
		    return false;
		}
		
		function _liteNormalize(t, cs) {
		    if (/^(pre|pre-wrap|break-spaces)$/.test(cs.whiteSpace)) t = t.replace(/[\r\n\t\u00A0]/g, " ");
		    else t = t.replace(/[\s\u00A0]+/g, " ");
		    if (cs.textTransform === "uppercase") t = t.toUpperCase();
		    else if (cs.textTransform === "lowercase") t = t.toLowerCase();
		    else if (cs.textTransform === "capitalize") t = t.replace(/(^|\s)(\S)/g, (m, a, b) => a + b.toUpperCase());
		    return t.replace(/[\u0000-\u001F\u007F\u200B-\u200F\u2028\u2029\uFEFF]/g, "");
		}
		
		// Tách 1 đoạn chữ nhiều dòng thành từng dòng (theo vị trí từng ký tự)
		function _liteSplitLines(node, range) {
		    const s = node.nodeValue;
		    const lines = [];
		    let cur = null;
		    for (let i = 0; i < s.length; i++) {
		        range.setStart(node, i);
		        range.setEnd(node, i + 1);
		        const r = range.getClientRects()[0];
		        if (!r || (r.width === 0 && r.height === 0)) { if (cur) cur.end = i + 1; continue; }
		        if (cur && Math.abs(r.top - cur.top) < r.height * 0.5) {
		            cur.end = i + 1;
		            cur.left = Math.min(cur.left, r.left);
		            cur.right = Math.max(cur.right, r.right);
		            cur.bottom = Math.max(cur.bottom, r.bottom);
		        } else {
		            cur = { start: i, end: i + 1, left: r.left, right: r.right, top: r.top, bottom: r.bottom };
		            lines.push(cur);
		        }
		    }
		    return lines.map((l) => ({ start: l.start, end: l.end, left: l.left, top: l.top, width: l.right - l.left, height: l.bottom - l.top }));
		}
		
		// Ảnh khác nguồn (CDN) mà <img> không có crossorigin sẽ làm canvas bị "tainted" (không xuất được).
		// Tải sẵn một bản có CORS để vẽ; nếu máy chủ không cho CORS thì bỏ qua (sẽ rơi về dự phòng).
		const _liteImgCache = new Map();
		
		function _isCrossOrigin(url) {
		    try {
		        if (/^(data|blob):/i.test(url)) return false;
		        return new URL(url, location.href).origin !== location.origin;
		    } catch (e) { return false; }
		}
		
		function preloadPageImages(div) {
		    const jobs = [];
		    div.querySelectorAll("img").forEach((img) => {
		        const src = img.currentSrc || img.src;
		        if (!src || img.crossOrigin === "anonymous" || !_isCrossOrigin(src) || _liteImgCache.has(img)) return;
		        jobs.push(new Promise((resolve) => {
		            const c = new Image();
		            c.crossOrigin = "anonymous";
		            c.onload = () => { _liteImgCache.set(img, c); resolve(); };
		            c.onerror = () => resolve();
		            c.src = src;
		        }));
		    });
		    return Promise.all(jobs);
		}
		
		function releasePageImages(div) {
		    div.querySelectorAll("img").forEach((img) => _liteImgCache.delete(img));
		}
		
		// mode: "strict" = vẽ, gặp thứ không hỗ trợ thì trả null
		//       "force"  = vẽ cố hết mức, bỏ qua thứ không hỗ trợ
		//       "scan"   = chỉ lấy lớp chữ, không vẽ
		function renderPageLite(div, opt, mode) {
		    const strict = mode === "strict";
		    const draw = mode !== "scan";
		    const pr = div.getBoundingClientRect();
		    if (pr.width < 1 || pr.height < 1) return null;
		
		    const k = opt.targetWidth / pr.width;
		    let canvas = null, ctx = null;
		    if (draw) {
		        canvas = document.createElement("canvas");
		        canvas.width = Math.round(opt.targetWidth);
		        canvas.height = Math.max(1, Math.round(pr.height * k));
		        ctx = canvas.getContext("2d");
		        ctx.fillStyle = "#ffffff";
		        ctx.fillRect(0, 0, canvas.width, canvas.height);
		    }
		
		    const runs = [];
		    const range = document.createRange();
		    const pageScale = pr.width / (div.offsetWidth || pr.width);
		    const scaleCache = new Map();
		    let failed = false;
		
		    const elScale = (el) => {
		        let s = scaleCache.get(el);
		        if (s === undefined) {
		            const w = el.offsetWidth;
		            const rw = el.getBoundingClientRect().width;
		            s = w > 0 && rw > 0 ? rw / w : pageScale;
		            scaleCache.set(el, s);
		        }
		        return s;
		    };
		
		    function drawRun(text, ln, el, cs) {
		        if (_colorAlpha(cs.color) === 0) return;
		        if (cs.webkitTextFillColor && _colorAlpha(cs.webkitTextFillColor) === 0) return;
		        const fsPx = parseFloat(cs.fontSize) * elScale(el) * k;
		        if (!(fsPx >= 1)) return;
		        ctx.font = cs.fontStyle + " " + cs.fontWeight + " " + fsPx + "px " + cs.fontFamily;
		        ctx.fillStyle = cs.color;
		        ctx.textBaseline = "alphabetic";
		        const w0 = ctx.measureText(text).width;
		        if (!(w0 > 0)) return;
		        const m = ctx.measureText("Hg");
		        const asc = m.fontBoundingBoxAscent !== undefined ? m.fontBoundingBoxAscent : fsPx * 0.8;
		        const desc = m.fontBoundingBoxDescent !== undefined ? m.fontBoundingBoxDescent : fsPx * 0.2;
		        const baseline = (ln.top - pr.top) * k + (ln.height * k - (asc + desc)) / 2 + asc;
		        ctx.save();
		        ctx.translate((ln.left - pr.left) * k, baseline);
		        ctx.scale((ln.width * k) / w0, 1); // ép chữ vừa đúng bề rộng của bản gốc
		        ctx.fillText(text, 0, 0);
		        ctx.restore();
		    }
		
		    function handleText(node, el, cs) {
		        if (cs.visibility === "hidden") return;
		        const raw = node.nodeValue;
		        if (!raw || !raw.trim()) return;
		        range.selectNodeContents(node);
		        const rects = range.getClientRects();
		        if (!rects.length) return;
		
		        let lines;
		        if (rects.length === 1) {
		            const r = rects[0];
		            lines = [{ start: 0, end: raw.length, left: r.left, top: r.top, width: r.width, height: r.height }];
		        } else if (raw.length > opt.maxCharsPerNode) {
		            if (strict) { failed = true; return; }
		            const r = range.getBoundingClientRect();
		            lines = [{ start: 0, end: raw.length, left: r.left, top: r.top, width: r.width, height: r.height }];
		        } else {
		            lines = _liteSplitLines(node, range);
		        }
		
		        for (const ln of lines) {
		            if (ln.width < 0.5 || ln.height < 0.5) continue;
		            const t = _liteNormalize(raw.slice(ln.start, ln.end), cs);
		            if (!t.trim()) continue;
		            runs.push({ text: t, x: ln.left - pr.left, y: ln.top - pr.top, w: ln.width, h: ln.height });
		            if (draw) drawRun(t, ln, el, cs);
		        }
		    }
		
		    function walk(el) {
		        if (failed) return;
		        const cs = getComputedStyle(el);
		        if (cs.display === "none") return;
		        if (parseFloat(cs.opacity) === 0) return;
		        const tag = el.tagName.toUpperCase();
		        if (strict && (LITE_BLOCK_TAGS.has(tag) || _liteUnsupported(el, cs))) { failed = true; return; }
		        const visible = cs.visibility !== "hidden";
		
		        if (draw && visible && _colorAlpha(cs.backgroundColor) > 0) {
		            const r = el.getBoundingClientRect();
		            ctx.fillStyle = cs.backgroundColor;
		            ctx.fillRect((r.left - pr.left) * k, (r.top - pr.top) * k, r.width * k, r.height * k);
		        }
		
		        if (tag === "IMG" || tag === "CANVAS") {
		            if (draw && visible) {
		                const okImg = tag === "IMG" ? (el.complete && el.naturalWidth > 0) : (el.width > 0 && el.height > 0);
		                if (!okImg || (tag === "IMG" && cs.objectFit !== "fill")) {
		                    if (strict) failed = true;
		                    return;
		                }
		                const r = el.getBoundingClientRect();
		                try {
		                    ctx.drawImage((tag === "IMG" && _liteImgCache.get(el)) || el, (r.left - pr.left) * k, (r.top - pr.top) * k, r.width * k, r.height * k);
		                } catch (e) {
		                    if (strict) failed = true;
		                }
		            }
		            return;
		        }
		
		        for (let n = el.firstChild; n; n = n.nextSibling) {
		            if (n.nodeType === 1) walk(n);
		            else if (n.nodeType === 3) handleText(n, el, cs);
		            if (failed) return;
		        }
		    }
		
		    walk(div);
		    if (failed) return null;
		    return { canvas, runs, pageW: pr.width, pageH: pr.height };
		}
		
		// Vẽ + mã hóa JPEG; trả null nếu không dùng được (để rơi về cách khác)
		async function liteItem(div, opt, mode) {
		    try {
		        const res = renderPageLite(div, opt, mode);
		        if (!res) return null;
		        const jpg = await canvasToJpeg(res.canvas, opt.quality);
		        if (!jpg) return null;
		        return { jpg: jpg, runs: res.runs, pageW: res.pageW, pageH: res.pageH };
		    } catch (e) {
		        console.warn("liteItem(" + mode + ") lỗi:", e.message);
		        return null; // ví dụ canvas bị chặn do ảnh khác nguồn
		    }
		}
		
		// ===== Chế độ 3: nhanh + copy được chữ =====
		function exportPdfText() {
		    const opt = TEXT_OPT;
		
		    return runExport(opt.preDelay, async () => {
		        const pages = getVisiblePages();
		        const total = pages.length;
		        if (total === 0) throw new Error("Không tìm thấy trang nào để xuất");
		
		        const landscape = /^l/i.test(String(khogiay));
		        const items = new Array(total);
		        const stats = {};
		        const state = { direct: FAST_OPT.direct, foreign: FAST_OPT.foreignObject };
		
		        // Ẩn tất cả trang, chỉ hiện từng trang khi xử lý (giúp bản clone của html2canvas nhỏ khi phải dự phòng)
		        const savedDisplay = pages.map((p) => p.style.display);
		        pages.forEach((p) => (p.style.display = "none"));
		
		        // Tải trước ảnh của vài trang kế tiếp (chỉ cần với ảnh khác nguồn)
		        const AHEAD = 3;
		        const pre = new Map();
		        const startPre = (j) => { if (j < total && !pre.has(j)) pre.set(j, preloadPageImages(pages[j])); };
		        for (let j = 0; j < Math.min(AHEAD, total); j++) startPre(j);
		
		        const tStart = performance.now();
		        try {
		            for (let i = 0; i < total; i++) {
		                const div = pages[i];
		                startPre(i + AHEAD);
		                await pre.get(i);
		                div.style.display = savedDisplay[i];
		                const t0 = performance.now();
		                let item = null;
		                let method = "lite";
		                try {
		                    item = await liteItem(div, opt, "strict");
		                    if (!item) {
		                        method = "fallback";
		                        if (opt.fallbackToHtml2canvas && typeof html2canvas === "function") {
		                            const r = await renderPageFast(div, FAST_OPT, state);
		                            const scan = renderPageLite(div, opt, "scan");
		                            if (r && scan) item = { jpg: r, runs: scan.runs, pageW: scan.pageW, pageH: scan.pageH };
		                        }
		                        if (!item) {
		                            method = "lite-force";
		                            item = await liteItem(div, opt, "force");
		                        }
		                    }
		                } finally {
		                    div.style.display = "none";
		                    releasePageImages(div);
		                    pre.delete(i);
		                }
		
		                items[i] = item;
		                stats[method] = (stats[method] || 0) + 1;
		                if (i === 0) showProgressOverlay();
		                updateProgress(Math.round(((i + 1) / total) * 100));
		                console.log("Trang " + i + ": " + method + " - " + Math.round(performance.now() - t0) + "ms");
		                await sleep(0);
		            }
		        } finally {
		            pages.forEach((p, i) => (p.style.display = savedDisplay[i]));
		        }
		        console.log("Cách xử lý mỗi trang:", stats, "| Tổng:", Math.round(performance.now() - tStart), "ms");
		
		        const valid = items.filter(Boolean);
		        if (valid.length === 0) throw new Error("Không render được trang nào");
		
		        downloadPdfBlob(buildTextPdf(valid, landscape));
		    });
		}
