		// Thêm 1 trang ảnh vào PDF theo kiểu "COVER": phủ kín trang A4, căn giữa
		function addCoverImage(pdf, imgData, ratio, alias, compression, isFirstPage) {
		    const pageWidth = pdf.internal.pageSize.getWidth();
		    const pageHeight = pdf.internal.pageSize.getHeight();
		    const pageRatio = pageWidth / pageHeight;
		
		    let imgWidth, imgHeight;
		    if (ratio > pageRatio) {
		        imgHeight = pageHeight;
		        imgWidth = pageHeight * ratio;
		    } else {
		        imgWidth = pageWidth;
		        imgHeight = pageWidth / ratio;
		    }
		    const x = (pageWidth - imgWidth) / 2;
		    const y = (pageHeight - imgHeight) / 2;
		
		    if (!isFirstPage) pdf.addPage();
		    pdf.addImage(imgData, "JPEG", x, y, imgWidth, imgHeight, alias, compression);
		}

		function exportPdfHighQuality() {
		    return runExport(HQ_OPT.preDelay, async () => {
		        const { jsPDF } = window.jspdf;
		        const pdf = new jsPDF(khogiay, "mm", "a4");
		
		        const pages = getVisiblePages();

		        if (total === 0) throw new Error("Không tìm thấy trang nào để xuất");
		
		        let added = 0;
		        for (let i = 0; i < total; i++) {
		            const div = pages[i];
		
		            const canvas = await html2canvas(div, {
		                scale: window.devicePixelRatio * 2,
		                useCORS: true,
		                backgroundColor: "#ffffff",
		                logging: false,
		                windowWidth: div.scrollWidth,
		                windowHeight: div.scrollHeight,
		            });
		
		            showProgressOverlay();
		
		            if (!canvas || canvas.width === 0 || canvas.height === 0) continue;
		
		            const imgData = canvas.toDataURL("image/jpeg", HQ_OPT.quality);
		            addCoverImage(pdf, imgData, canvas.width / canvas.height, undefined, HQ_OPT.compression, added === 0);
		
		            added++;
		            updateProgress(Math.round((added / total) * 100));
		
		            // Nhường luồng cho trình duyệt cập nhật UI
		            await sleep(0);
		        }
		
		        if (added === 0) throw new Error("Không render được trang nào");
		
		        await sleep(HQ_OPT.holdDelay);
		        downloadPdf(pdf);
		    });
		}

		// canvas -> { data: Uint8Array (JPEG), ratio } ; giải phóng canvas sau khi dùng
		async function canvasToJpeg(canvas, quality) {
		    if (!canvas || canvas.width === 0 || canvas.height === 0) return null;
		    const ratio = canvas.width / canvas.height;
		    // toBlob mã hóa bất đồng bộ, nhanh và nhẹ hơn toDataURL (base64)
		    const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", quality));
		    canvas.width = canvas.height = 0; // trả bộ nhớ sớm
		    if (!blob) return null;
		    return { data: new Uint8Array(await blob.arrayBuffer()), ratio: ratio };
		}
		
		// Kiểm tra canvas có phải trang trắng toàn bộ không (dấu hiệu foreignObject render hỏng).
		// Ném lỗi nếu canvas bị "tainted" (ảnh khác nguồn) -> người gọi sẽ rơi về tầng sau.
		function isBlankCanvas(canvas) {
		    const S = 24;
		    const small = document.createElement("canvas");
		    small.width = small.height = S;
		    const ctx = small.getContext("2d", { willReadFrequently: true });
		    ctx.drawImage(canvas, 0, 0, S, S);
		    const d = ctx.getImageData(0, 0, S, S).data;
		    for (let i = 0; i < d.length; i += 4) {
		        if (d[i + 3] < 10) continue; // pixel trong suốt
		        if (d[i] < 250 || d[i + 1] < 250 || d[i + 2] < 250) return false;
		    }
		    return true;
		}
		
		// Tầng 1: trang chỉ là 1 ảnh/canvas phủ gần kín -> lấy thẳng, không cần "chụp" DOM.
		// Trả null nếu trang không đủ điều kiện (để không làm mất nội dung).
		async function tryDirectImage(div, opt) {
		    if (div.textContent.trim() !== "") return null;                                // có chữ -> không dùng
		    if (div.querySelector("svg, video, iframe, object, embed")) return null;       // có thành phần khác
		    const medias = div.querySelectorAll("img, canvas");
		    if (medias.length !== 1) return null;
		    const el = medias[0];
		
		    // Có ảnh nền ở đâu đó thì nội dung đó sẽ bị mất -> không dùng
		    const all = [div, ...div.querySelectorAll("*")];
		    if (all.some((n) => getComputedStyle(n).backgroundImage !== "none")) return null;
		
		    const d = div.getBoundingClientRect();
		    const r = el.getBoundingClientRect();
		    if (d.width < 1 || d.height < 1) return null;
		    if (r.width * r.height < 0.9 * d.width * d.height) return null;                // ảnh không phủ gần kín trang
		
		    if (el.tagName === "IMG") {
		        if (!(el.complete && el.naturalWidth > 0)) return null;
		        const fit = getComputedStyle(el).objectFit;
		        if (fit && fit !== "fill") return null;
		    } else if (el.width === 0 || el.height === 0) {
		        return null;
		    }
		
		    const c = document.createElement("canvas");
		    c.width = Math.round(opt.targetWidth);
		    c.height = Math.max(1, Math.round((opt.targetWidth * d.height) / d.width));
		    const sx = c.width / d.width;
		    const ctx = c.getContext("2d");
		    ctx.fillStyle = "#ffffff";
		    ctx.fillRect(0, 0, c.width, c.height);
		    ctx.imageSmoothingQuality = "high";
		    ctx.drawImage(el, (r.left - d.left) * sx, (r.top - d.top) * sx, r.width * sx, r.height * sx);
		
		    // Ném SecurityError nếu ảnh khác nguồn không có CORS -> người gọi sẽ bắt lại
		    return await canvasToJpeg(c, opt.quality);
		}
		
		// Render 1 trang theo thứ tự: direct -> foreignObject -> html2canvas thường
		// state: cờ dùng chung, tầng nào đã lỗi thì các trang sau bỏ qua luôn
		async function renderPageFast(div, opt, state) {
		    // --- Tầng 1 ---
		    if (state.direct) {
		        try {
		            const r = await tryDirectImage(div, opt);
		            if (r) return { ...r, method: "direct" };
		        } catch (e) {
		            state.direct = false;
		        }
		    }
		
		    const w = div.offsetWidth || div.scrollWidth || 1;
		    const scale = Math.min(Math.max(opt.targetWidth / w, opt.minScale), opt.maxScale);
		    const baseOpts = {
		        scale: scale,
		        useCORS: true,
		        backgroundColor: "#ffffff",
		        logging: false,
		        windowWidth: div.scrollWidth,
		        windowHeight: div.scrollHeight,
		    };
		
		    // --- Tầng 2 ---
		    if (state.foreign) {
		        try {
		            const canvas = await html2canvas(div, { ...baseOpts, foreignObjectRendering: true });
		            if (!isBlankCanvas(canvas)) {
		                const r = await canvasToJpeg(canvas, opt.quality);
		                if (r) return { ...r, method: "foreignObject" };
		            }
		            state.foreign = false; // ra trang trắng -> các trang sau khỏi thử lại
		        } catch (e) {
		            state.foreign = false;
		        }
		    }
		
		    // --- Tầng 3 ---
		    const canvas = await html2canvas(div, { ...baseOpts, foreignObjectRendering: false });
		    const r = await canvasToJpeg(canvas, opt.quality);
		    return r ? { ...r, method: "standard" } : null;
		}
		
		// ===== Chế độ 1: bình thường (nhanh) =====
		async function exportPdfFast() {
		    const opt = FAST_OPT;
		    let needPrint = false;
		
		    await runExport(opt.preDelay, async () => {
		        const { jsPDF } = window.jspdf;
		        const pdf = new jsPDF(khogiay, "mm", "a4");
		
		        const pages = getVisiblePages();
		        const total = pages.length;
		        if (total === 0) throw new Error("Không tìm thấy trang nào để xuất");
		
		        const results = new Array(total); // { data: Uint8Array, ratio, method }
		        const state = { direct: opt.direct, foreign: opt.foreignObject };
		        const methodCount = {};
		
		        // Ẩn tất cả trang, chỉ hiện từng trang một khi chụp.
		        // html2canvas clone cả tài liệu cho mỗi lần chụp, nên ẩn bớt giúp bản clone nhỏ hơn nhiều.
		        const savedDisplay = pages.map((p) => p.style.display);
		        pages.forEach((p) => (p.style.display = "none"));
		
		        const tStart = performance.now();
		        try {
		            for (let i = 0; i < total; i++) {
		                const t0 = performance.now();
		
		                pages[i].style.display = savedDisplay[i];
		                let r = null;
		                try {
		                    r = await renderPageFast(pages[i], opt, state);
		                } finally {
		                    pages[i].style.display = "none";
		                }
		                const dt = performance.now() - t0;
		
		                if (i === 0) showProgressOverlay();
		                if (r) {
		                    results[i] = r;
		                    methodCount[r.method] = (methodCount[r.method] || 0) + 1;
		                }
		
		                // Sau trang đầu: nếu ước tính quá lâu thì đề xuất chuyển sang chế độ in
		                if (i === 0 && opt.suggestPrint && r && r.method !== "direct" && dt * total > opt.slowEstimateMs) {
		                    const sec = Math.round((dt * total) / 1000);
		                    const ok = confirm(
		                        "Cách chụp ảnh với tài liệu này khá chậm (ước tính khoảng " + sec + " giây).\n\n" +
		                        "Chuyển sang chế độ in bằng trình duyệt (nhanh hơn nhiều, bạn chọn 'Save as PDF')?\n\n" +
		                        "OK = chuyển sang chế độ in | Cancel = tiếp tục chụp ảnh"
		                    );
		                    if (ok) {
		                        needPrint = true;
		                        return; // finally bên dưới sẽ trả lại hiển thị các trang
		                    }
		                }
		
		                updateProgress(Math.round(((i + 1) / total) * 100));
		                console.log(`Trang ${i}: ${r ? r.method : "lỗi"} - ${Math.round(dt)}ms`);
		                await sleep(0);
		            }
		        } finally {
		            // Luôn trả lại trạng thái hiển thị ban đầu của các trang
		            pages.forEach((p, i) => (p.style.display = savedDisplay[i]));
		        }
		        console.log("Cách render mỗi trang:", methodCount, "| Tổng:", Math.round(performance.now() - tStart), "ms");
		
		        // Ghép PDF theo đúng thứ tự trang
		        let added = 0;
		        for (let i = 0; i < total; i++) {
		            const r = results[i];
		            if (!r) continue;
		            // alias duy nhất giúp jsPDF bỏ qua bước băm dữ liệu ảnh để tìm ảnh trùng
		            addCoverImage(pdf, r.data, r.ratio, "p" + i, "NONE", added === 0);
		            results[i] = null;
		            added++;
		        }
		        if (added === 0) throw new Error("Không render được trang nào");
		
		        downloadPdf(pdf);
		    });
		
		    if (needPrint) return exportPdfPrint();
		}
		
		// ===== Chế độ 2: in bằng trình duyệt (nhanh với mọi loại trang) =====
		// Dựng iframe ẩn chỉ chứa các trang (đã co vừa A4) rồi gọi print().
		// Người dùng chọn "Save as PDF" trong hộp thoại in.
		function exportPdfPrint() {
		    return runExport(0, async () => {
		        const pages = getVisiblePages();
		        if (pages.length === 0) throw new Error("Không tìm thấy trang nào để xuất");
		
		        const landscape = /^l/i.test(String(khogiay));
		        const A4W = landscape ? 1122.5 : 793.7; // px ở 96dpi
		        const A4H = landscape ? 793.7 : 1122.5;
		        const mmW = landscape ? 297 : 210;
		        const mmH = landscape ? 210 : 297;
		
		        // Xóa iframe của lần trước (nếu có), tạo iframe mới
		        if (window.__printFrame) window.__printFrame.remove();
		        const iframe = document.createElement("iframe");
		        iframe.style.cssText = `position:fixed;left:-10000px;top:0;width:${A4W}px;height:${A4H}px;border:0;`;
		        document.body.appendChild(iframe);
		        window.__printFrame = iframe;
		
		        const doc = iframe.contentDocument;
		        doc.open();
		        doc.write(`<!DOCTYPE html><html><head><meta charset="utf-8"><base href="${document.baseURI}"></head><body></body></html>`);
		        doc.close();
		
		        // Sao chép CSS của trang (đọc từ CSSOM để lấy cả style được thêm bằng JS)
		        for (const sheet of Array.from(document.styleSheets)) {
		            try {
		                const css = Array.from(sheet.cssRules).map((r) => r.cssText).join("\n");
		                const st = doc.createElement("style");
		                st.textContent = css;
		                doc.head.appendChild(st);
		            } catch (e) {
		                // Stylesheet khác nguồn không đọc được rule -> nạp lại bằng link
		                if (sheet.href) {
		                    const l = doc.createElement("link");
		                    l.rel = "stylesheet";
		                    l.href = sheet.href;
		                    doc.head.appendChild(l);
		                }
		            }
		        }
		
		        // CSS dành cho bản in (đặt sau cùng để ghi đè)
		        const printStyle = doc.createElement("style");
		        printStyle.textContent = `
		            @page { size: A4 ${landscape ? "landscape" : "portrait"}; margin: 0; }
		            html, body { margin: 0; padding: 0; background: #fff; }
		            * { -webkit-print-color-adjust: exact; print-color-adjust: exact; }
		            .pp { width: ${mmW}mm; height: ${mmH - 0.5}mm; overflow: hidden; display: flex;
		                  align-items: center; justify-content: center;
		                  break-after: page; page-break-after: always; }
		            .pp:last-child { break-after: auto; page-break-after: auto; }
		        `;
		        doc.head.appendChild(printStyle);
		
		        // Dựng các trang: mỗi trang co vừa A4 (zoom) và căn giữa
		        const container = $("page-container");
		        const wrap = doc.createElement("div");
		        wrap.id = "page-container";
		        wrap.className = container.className;
		        doc.body.appendChild(wrap);
		
		        for (const page of pages) {
		            const W = page.offsetWidth || 1;
		            const H = page.offsetHeight || 1;
		            const z = Math.min(A4W / W, A4H / H);
		
		            const clone = page.cloneNode(true);
		
		            // canvas khi clone sẽ bị rỗng -> chép lại nội dung
		            const srcCv = page.querySelectorAll("canvas");
		            const dstCv = clone.querySelectorAll("canvas");
		            srcCv.forEach((c, k) => {
		                const d = dstCv[k];
		                if (!d || c.width === 0 || c.height === 0) return;
		                d.width = c.width;
		                d.height = c.height;
		                d.getContext("2d").drawImage(c, 0, 0);
		            });
		
		            Object.assign(clone.style, {
		                position: "relative",
		                left: "auto", top: "auto", right: "auto", bottom: "auto",
		                transform: "none",
		                margin: "0",
		                flex: "none",
		                width: W + "px",
		                height: H + "px",
		                zoom: String(z),
		            });
		
		            const sheetDiv = doc.createElement("div");
		            sheetDiv.className = "pp";
		            sheetDiv.appendChild(clone);
		            wrap.appendChild(sheetDiv);
		        }
		
		        // Chờ ảnh và font tải xong
		        const imgs = Array.from(doc.images);
		        imgs.forEach((img) => (img.loading = "eager"));
		        await Promise.all(imgs.map((img) =>
		            img.complete ? null : new Promise((res) => {
		                img.onload = img.onerror = res;
		                setTimeout(res, 8000);
		            })
		        ));
		        if (doc.fonts && doc.fonts.ready) await Promise.race([doc.fonts.ready, sleep(3000)]);
		        await sleep(150);
		
		        // Mở hộp thoại in (Chrome sẽ chờ ở đây cho đến khi đóng hộp thoại)
		        iframe.contentWindow.focus();
		        iframe.contentWindow.print();
		    });
		}
		
		// ===== Gắn sự kiện =====
		$("btnExport").onclick = function () {
		    const mode = $("slOption").selectedIndex;
		    if (mode === 2) return exportPdfPrint();      // In nhanh bằng trình duyệt
		    if (mode === 1) return exportPdfFast();       // Bình thường (nhanh)
		    return exportPdfHighQuality();                // Chất lượng cao
		};
