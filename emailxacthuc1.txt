  <script>
      (function () {
          var MAX_LEN = 150;
          var modal = document.getElementById('modalBoSung');
          var txtModal = document.getElementById('txtBoSungModal');
          var txtHidden = document.getElementById('txtBoSung');
          var counter = document.getElementById('txtBoSungCounter');

          // MỚI (v2): cập nhật bộ đếm
          function updateCounter() {
              var len = txtModal.value.length;
              counter.textContent = len + '/' + MAX_LEN;
              if (len >= MAX_LEN) {
                  counter.classList.add('full');
              } else {
                  counter.classList.remove('full');
              }
          }

          // MỚI (v2): đổi mọi dấu xuống dòng thành khoảng trắng (khi paste)
          function stripNewLines() {
              if (/[\r\n]/.test(txtModal.value)) {
                  txtModal.value = txtModal.value.replace(/[\r\n]+/g, ' ').substring(0, MAX_LEN);
              }
          }

          // MỚI (v2): chặn phím Enter
          txtModal.addEventListener('keydown', function (e) {
              if (e.key === 'Enter' || e.keyCode === 13) {
                  e.preventDefault();
              }
          });

          txtModal.addEventListener('input', function () {
              stripNewLines();
              updateCounter();
          });

          function openModal() {
              txtModal.value = txtHidden.value;
              updateCounter();
              modal.classList.add('show');
              txtModal.focus();
          }

          function closeModal() {
              modal.classList.remove('show');
          }

          document.getElementById('btnOpenModal').addEventListener('click', openModal);
          document.getElementById('btnCloseModal').addEventListener('click', closeModal);

          document.getElementById('btnBoSung').addEventListener('click', function () {
              stripNewLines();
              txtHidden.value = txtModal.value;
              closeModal();
          });
      })();
  </script>
